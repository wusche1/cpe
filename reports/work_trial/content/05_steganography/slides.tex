\subsection{Steganographic reasoning}
\coverslide{Steganographic reasoning}

\begin{frame}{The steganographic-reasoning organism}
  \begin{itemize}
    \item Gets a main task (an ordinary user request) and a side task (track a ball under shuffled cups).
    \item Hides the side-task reasoning inside the main-task answer, in the tokenization.
    \item Answers the main task, then gives the side task's solution.
  \end{itemize}
\end{frame}
